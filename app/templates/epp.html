document.getElementById("btn-guardar-epp").addEventListener("click", async function(e) { e.preventDefault(); let formData = new FormData(); formData.append("cedula", document.querySelector("input[placeholder='Digite cédula...']").value); formData.append("nombre", document.querySelector("input[placeholder='Nombre completo']").value); formData.append("cargo", document.querySelector("input[placeholder='Cargo']").value); formData.append("area", document.querySelector("input[placeholder='Área']").value); formData.append("fecha_ingreso", document.querySelector("input[type='date']").value); // Adjungere firmas e canvas formData.append("firma_colaborador", canvasColaborador.toDataURL()); formData.append("firma_entrega", canvasEntrega.toDataURL()); try { let response = await fetch("/epp/guardar", { method: "POST", body: formData }); if (!response.ok) { throw new Error("Ruta no encontrada o error en servidor (" + response.status + ")"); } let result = await response.json(); if (result.status === "success") { alert("✅ Registro guardado con éxito."); window.open(result.pdf_url, "_blank"); // Aperit PDF in nova tabula } else { alert("❌ Error: " + result.message); } } catch (err) { alert("❌ Error: " + err.message); } });
